const SHEET_NAME = 'KetQua';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const sheet = getOrCreateSheet();

    sheet.appendRow([
      new Date(),                 // Thời gian nộp bài (giờ server)
      data.name || '',            // Họ tên học viên
      data.testId || '',          // Mã bài kiểm tra (vd: KT1-5)
      data.testTitle || '',       // Tên bài kiểm tra
      data.score != null ? data.score : '',   // Điểm đúng
      data.total != null ? data.total : '',   // Tổng số câu
      data.percent != null ? data.percent : '', // Phần trăm đúng
      data.timeSeconds != null ? data.timeSeconds : '', // Thời gian làm bài (giây)
      data.date || ''             // Thời điểm nộp bài (giờ trình duyệt học viên)
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({status: 'ok'}))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({status: 'error', message: err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      'Thời gian nộp (server)', 'Họ tên học viên', 'Mã bài KT', 'Tên bài KT',
      'Điểm đúng', 'Tổng số câu', 'Phần trăm (%)', 'Thời gian làm (giây)', 'Thời gian nộp (trình duyệt)'
    ]);
    sheet.getRange(1, 1, 1, 9).setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

   chọn hàm này trong dropdown "Select function" rồi bấm Run,
   nếu chạy không lỗi và thấy dòng mới xuất hiện trong sheet "KetQua" là đã hoạt động đúng. */
function testDoPost() {
  const fakeEvent = {
    postData: {
      contents: JSON.stringify({
        testId: 'KT1-5', testTitle: 'Kiểm tra Bài 1–5 (test)',
        name: 'Học viên thử nghiệm', score: 8, total: 10, percent: 80,
        timeSeconds: 125, date: new Date().toISOString()
      })
    }
  };
  doPost(fakeEvent);
}
